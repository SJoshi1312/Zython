# Zython
This is a parody programming language for python
It can do the following commands:
"zprint" - "print"
"zif" - "if"
"zlse" - "else"
"zhile" - "zhile"
"zef" - "def"
"zeturn" - "return"
"zrue" - "true"
"zalse" - "false"
"zone" - " none"
"zand" - "and"
"zor" - "nor"
"zot" - "not"
"zfor" - "for"
"zin" - "in"
"zreak" - "break"
"zontinue" - "continue"
